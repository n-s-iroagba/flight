Data Merging Strategy
Search Execution:

Admin flights queried from local MySQL database

LetsFG API called for real-time flight data

Both sources searched simultaneously for performance

Merging Logic:

Results deduplicated by flightNumber + departureDate

Priority: Admin flights shown first (if featured)

All results sorted by price (lowest to highest)

Each result tagged with source: 'admin' | 'letsfg'

Caching Strategy:

LetsFG results cached for 5 minutes (TTL)

Admin flights cached for 1 hour (TTL)

Featured flights cached for 1 hour (TTL)

Database Schema
sql
-- Flights table
CREATE TABLE flights (
  id VARCHAR(36) PRIMARY KEY,
  origin VARCHAR(3) NOT NULL,
  destination VARCHAR(3) NOT NULL,
  departure_time DATETIME NOT NULL,
  arrival_time DATETIME NOT NULL,
  airline VARCHAR(100) NOT NULL,
  airline_code VARCHAR(2) NOT NULL,
  flight_number VARCHAR(10) NOT NULL,
  price DECIMAL(10,2) NOT NULL,
  currency VARCHAR(3) DEFAULT 'USD',
  total_seats INT NOT NULL,
  available_seats INT NOT NULL,
  aircraft VARCHAR(100),
  baggage TEXT,
  cancellation_policy TEXT,
  cabin_class ENUM('economy', 'premium_economy', 'business', 'first'),
  status ENUM('active', 'inactive', 'sold_out') DEFAULT 'active',
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  INDEX idx_route (origin, destination),
  INDEX idx_departure (departure_time),
  INDEX idx_status (status)
);

-- Bookings table
CREATE TABLE bookings (
  id VARCHAR(36) PRIMARY KEY,
  booking_reference VARCHAR(20) UNIQUE NOT NULL,
  flight_id VARCHAR(36) NOT NULL,
  passenger_name VARCHAR(255) NOT NULL,
  passenger_email VARCHAR(255) NOT NULL,
  passenger_phone VARCHAR(20) NOT NULL,
  special_requests TEXT,
  source ENUM('admin', 'letsfg') NOT NULL,
  offer_id VARCHAR(100),
  amount DECIMAL(10,2) NOT NULL,
  currency VARCHAR(3) DEFAULT 'USD',
  status ENUM('pending', 'processing', 'paid', 'failed', 'cancelled') DEFAULT 'pending',
  whatsapp_conversation TEXT,
  paid_at TIMESTAMP NULL,
  confirmed_by VARCHAR(100),
  notes TEXT,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  expires_at TIMESTAMP DEFAULT (CURRENT_TIMESTAMP + INTERVAL 30 MINUTE),
  FOREIGN KEY (flight_id) REFERENCES flights(id),
  INDEX idx_reference (booking_reference),
  INDEX idx_status (status),
  INDEX idx_created (created_at)
);

-- Payments table
CREATE TABLE payments (
  id VARCHAR(36) PRIMARY KEY,
  booking_id VARCHAR(36) NOT NULL,
  amount DECIMAL(10,2) NOT NULL,
  currency VARCHAR(3) DEFAULT 'USD',
  method ENUM('cash', 'bank_transfer', 'mobile_money', 'whatsapp') NOT NULL,
  reference VARCHAR(100),
  status ENUM('pending', 'completed', 'failed') DEFAULT 'pending',
  confirmed_by VARCHAR(100),
  confirmed_at TIMESTAMP NULL,
  notes TEXT,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (booking_id) REFERENCES bookings(id),
  INDEX idx_booking (booking_id),
  INDEX idx_status (status)
);

-- Tickets table
CREATE TABLE tickets (
  id VARCHAR(36) PRIMARY KEY,
  booking_id VARCHAR(36) NOT NULL,
  ticket_number VARCHAR(50) UNIQUE NOT NULL,
  pnr VARCHAR(10) NOT NULL,
  e_ticket_url VARCHAR(255),
  qr_code TEXT,
  delivery_status ENUM('pending', 'sent', 'confirmed') DEFAULT 'pending',
  sent_via ENUM('whatsapp', 'email'),
  sent_at TIMESTAMP NULL,
  confirmed_at TIMESTAMP NULL,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (booking_id) REFERENCES bookings(id),
  INDEX idx_ticket (ticket_number),
  INDEX idx_pnr (pnr)
);

-- Admin users table
CREATE TABLE admin_users (
  id VARCHAR(36) PRIMARY KEY,
  email VARCHAR(255) UNIQUE NOT NULL,
  password_hash VARCHAR(255) NOT NULL,
  full_name VARCHAR(255) NOT NULL,
  role ENUM('admin', 'super_admin') DEFAULT 'admin',
  last_login TIMESTAMP NULL,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  INDEX idx_email (email)
);
